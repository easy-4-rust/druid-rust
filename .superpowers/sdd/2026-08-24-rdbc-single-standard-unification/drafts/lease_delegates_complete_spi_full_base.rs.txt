{
    let drop_count = Arc::new(AtomicUsize::new(0));
    let delegation_counts = Arc::new(DelegationCounts::default());
    let delegation_resources = delegation_resources();
    let mut connection = connection_lease_with_delegation(
        drop_count.clone(),
        Arc::clone(&delegation_counts),
        Arc::clone(&delegation_resources),
    );

    let statement = connection
        .create_physical_statement(PhysicalStatementOptions::default())
        .await
        .expect("physical statement must delegate");
    assert!(Arc::ptr_eq(&statement, &delegation_resources.statement));
    let prepared_key = PreparedStatementKey::new(
        Some("UPDATE delegated prepared".to_string()),
        None,
        PreparedStatementMethodType::M1,
    )
    .expect("prepared key must build");
    let delegated_prepared = connection
        .prepare_physical_statement(&prepared_key)
        .await
        .expect("physical prepare must delegate");
    assert!(Arc::ptr_eq(
        &delegated_prepared,
        &delegation_resources.prepared_statement
    ));
    let call_key = PreparedStatementKey::new(
        Some("CALL delegated".to_string()),
        None,
        PreparedStatementMethodType::M1,
    )
    .expect("call key must build");
    let callable = connection
        .prepare_physical_call(&call_key)
        .await
        .expect("physical call must delegate");
    assert!(Arc::ptr_eq(
        &callable,
        &delegation_resources.callable_statement
    ));
    assert_eq!(
        connection
            .exec_batch(vec![(
                "UPDATE delegated batch".to_string(),
                vec![Value::Int(11)]
            )])
            .await
            .expect("batch must delegate"),
        vec![11]
    );
    assert_eq!(
        connection
            .exec_prepared(delegated_prepared.as_ref(), vec![Value::Int(12)])
            .await
            .expect("prepared exec must delegate")
            .rows_affected,
        12
    );
    assert_eq!(
        connection
            .fetch_prepared(delegated_prepared.as_ref(), vec![Value::Int(13)])
            .await
            .expect("prepared fetch must delegate")[0]
            .values,
        vec![Value::Int(13)]
    );
    let mut metadata = connection
        .database_meta_data()
        .expect("metadata must delegate");
    assert_eq!(
        metadata
            .get_database_product_name()
            .await
            .expect("metadata value must delegate"),
        Some("lease-metadata".to_string())
    );
    drop(metadata);
    assert_eq!(
        connection.create_blob().await.expect("blob must delegate"),
        delegation_resources.blob
    );
    assert_eq!(
        connection.create_clob().await.expect("clob must delegate"),
        delegation_resources.clob
    );
    assert_eq!(
        connection
            .create_n_clob()
            .await
            .expect("nclob must delegate"),
        delegation_resources.n_clob
    );
    assert_eq!(
        delegation_counts
            .create_physical_statement
            .load(Ordering::Relaxed),
        1
    );
    assert_eq!(
        delegation_counts
            .prepare_physical_statement
            .load(Ordering::Relaxed),
        1
    );
    assert_eq!(
        delegation_counts
            .prepare_physical_call
            .load(Ordering::Relaxed),
        1
    );
    assert_eq!(delegation_counts.exec_batch.load(Ordering::Relaxed), 1);
    assert_eq!(delegation_counts.exec_prepared.load(Ordering::Relaxed), 1);
    assert_eq!(delegation_counts.fetch_prepared.load(Ordering::Relaxed), 1);
    assert_eq!(
        delegation_counts.database_meta_data.load(Ordering::Relaxed),
        1
    );
    assert_eq!(delegation_counts.create_blob.load(Ordering::Relaxed), 1);
    assert_eq!(delegation_counts.create_clob.load(Ordering::Relaxed), 1);
    assert_eq!(delegation_counts.create_n_clob.load(Ordering::Relaxed), 1);

    let exec_result = connection
        .exec("UPDATE contract", vec![Value::Int(7)])
        .await
        .expect("exec must delegate");
    assert_eq!(exec_result.rows_affected, 1);
    let execute_results = connection
        .execute(
            "EXECUTE contract",
            vec![Value::Int(8)],
            StatementGeneratedKeys::AutoGeneratedKeys(1),
        )
        .await
        .expect("generic execute must delegate");
    assert_eq!(execute_results[0].update_count(), 3);
    assert!(!execute_results[0].is_result_set());
    let rows = connection
        .fetch("SELECT contract", vec![Value::String("value".to_string())])
        .await
        .expect("fetch must delegate");
    assert_eq!(rows[0].values, vec![Value::Bool(true)]);
    let result_set = connection
        .fetch_result_set("SELECT contract", vec![Value::String("value".to_string())])
        .await
        .expect("ResultSet fetch must delegate");
    assert!(result_set.next().unwrap());
    assert!(result_set.boolean(1).unwrap());
    let query_prepared: Arc<dyn PhysicalPreparedStatement> =
        Arc::new(SqlTextPreparedStatement::new("SELECT contract"));
    let prepared_result_set = connection
        .fetch_prepared_result_set(
            query_prepared.as_ref(),
            vec![Value::String("value".to_string())],
        )
        .await
        .expect("prepared ResultSet fetch must delegate");
    assert!(prepared_result_set.next().unwrap());
    assert!(prepared_result_set.boolean(1).unwrap());
    let generic_prepared: Arc<dyn PhysicalPreparedStatement> =
        Arc::new(SqlTextPreparedStatement::new("EXECUTE contract"));
    let prepared_results = connection
        .execute_prepared(
            generic_prepared.as_ref(),
            vec![Value::Int(8)],
            StatementGeneratedKeys::AutoGeneratedKeys(1),
        )
        .await
        .expect("prepared generic execute must delegate");
    assert_eq!(prepared_results[0].update_count(), 3);
    let prepared: Arc<dyn PhysicalPreparedStatement> =
        Arc::new(SqlTextPreparedStatement::new("UPDATE contract"));
    assert_eq!(
        connection
            .exec_prepared_batch(prepared.as_ref(), vec![vec![Value::Int(7)]])
            .await
            .expect("prepared batch must delegate"),
        [1]
    );
    connection
        .close_prepared_statement(Arc::clone(&prepared))
        .await
        .expect("prepared close must delegate");
    assert!(prepared.is_closed());

    connection.begin().await.expect("begin must delegate");
    assert!(!connection.auto_commit());
    connection.rollback().await.expect("rollback must delegate");
    connection
        .begin()
        .await
        .expect("second begin must delegate");
    connection.commit().await.expect("commit must delegate");
    connection
        .set_auto_commit(false)
        .await
        .expect("set auto commit must delegate");
    connection
        .set_read_only(true)
        .await
        .expect("set read only must delegate");
    connection
        .set_transaction_isolation(8)
        .await
        .expect("set isolation must delegate");
    connection
        .set_holdability(1)
        .await
        .expect("set holdability must delegate");
    assert_eq!(
        connection
            .warnings()
            .await
            .expect("get warnings must delegate")
            .expect("warning must exist")
            .message(),
        "lease-warning"
    );
    connection
        .clear_warnings()
        .await
        .expect("clear warnings must delegate");
    connection
        .set_catalog("catalog_b")
        .await
        .expect("set catalog must delegate");
    connection
        .set_schema("schema_b")
        .await
        .expect("set schema must delegate");
    assert!(!connection.auto_commit());
    assert!(connection.read_only());
    assert_eq!(connection.transaction_isolation(), 8);
    assert_eq!(connection.holdability(), 1);
    assert_eq!(connection.catalog(), Some("catalog_b"));
    assert_eq!(connection.schema(), Some("schema_b"));

    let anonymous = connection
        .set_savepoint()
        .await
        .expect("anonymous savepoint must delegate");
    connection
        .rollback_to(&anonymous)
        .await
        .expect("rollback to savepoint must delegate");
    let named = connection
        .set_savepoint_named("named")
        .await
        .expect("named savepoint must delegate");
    connection
        .release_savepoint(&named)
        .await
        .expect("release savepoint must delegate");

    connection.ping().await.expect("ping must delegate");
    assert_eq!(connection.driver_name(), "lease-contract");
    assert_eq!(
        connection.capabilities(),
        PhysicalConnectionCapabilities {
            transactions: true,
            savepoints: true,
            auto_commit: true,
            read_only: true,
            transaction_isolation: true,
            holdability: true,
            clear_warnings: true,
            catalog: true,
            schema: true,
        }
    );
    assert!(!connection.is_discarded());
    connection.mark_discarded();
    assert!(connection.is_discarded());
    connection.abort().await.expect("abort must delegate");
    assert!(connection.is_closed());
    connection.close().await.expect("close must delegate");

    drop(connection);
    assert_eq!(drop_count.load(Ordering::Relaxed), 1);
}
