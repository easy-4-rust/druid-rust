
mod ordered_navigation_property_filter;

/// 防止短路仍移动原生游标、顺序颠倒或一次失败重复记错。
#[tokio::test]
async fn task6_navigation_property_order_short_circuit_and_error_counts() {
    use crate::core::{FilterChain, RdbcError};
    use druid_rdbc::result_set::{ResultSetNavigation, ResultSetExt};
    use ordered_navigation_property_filter::OrderedNavigationPropertyFilter;
    use std::sync::Mutex;
    let factory = ToastyConnectionFactory::new("sqlite::memory:").await.unwrap();
    for mode in [0, 1, 2, 3] {
        let calls = Arc::new(Mutex::new(Vec::new()));
        let mut filters = FilterChain::new();
        for name in ["A", "B", "C"] {
            filters.add_result_set(Arc::new(OrderedNavigationPropertyFilter {
                name, calls: Arc::clone(&calls), mode: if name == "B" {mode} else {0},
            }));
        }
        let (mut statement, mut connection) = task6_sqlite_metadata_statement(&factory, 180+u64::from(mode), "navigation-controls", Arc::new(filters)).await;
        let mut rows = statement.execute_query("SELECT 10 AS value UNION ALL SELECT 20").await.unwrap();
        let navigation = rows.navigate(ResultSetNavigation::Last).await;
        if mode == 2 {
            assert_eq!(navigation, Err(RdbcError::DriverError("ordered navigation failure".to_owned())));
            assert_eq!(*calls.lock().unwrap(), ["A:nav:Last:before", "B:nav:Last:before", "B:nav:error", "A:nav:error"]);
            assert_eq!(statement.exception_count(), 1);
        } else {
            assert_eq!(navigation, Ok(true));
            let expected: &[&str] = if mode == 1 {
                &["A:nav:Last:before", "B:nav:Last:before", "B:nav:ok", "A:nav:ok"]
            } else {
                &["A:nav:Last:before", "B:nav:Last:before", "C:nav:Last:before", "C:nav:ok", "B:nav:ok", "A:nav:ok"]
            };
            assert_eq!(*calls.lock().unwrap(), expected);
        }
        calls.lock().unwrap().clear();
        let property = rows.row();
        match mode {
            0 => assert_eq!(property, Ok(2)),
            1 => assert_eq!(property, Ok(77)),
            2 => assert_eq!(property, Err(RdbcError::DriverError("ordered property failure".to_owned()))),
            3 => assert_eq!(property, Err(RdbcError::InvalidArgument("incompatible result set property filter value".to_owned()))),
            _ => unreachable!(),
        }
        let expected: &[&str] = match mode {
            0 => &["A:property:Row:before", "B:property:Row:before", "C:property:Row:before", "C:property:ok", "B:property:ok", "A:property:ok"],
            2 => &["A:property:Row:before", "B:property:Row:before", "B:property:error", "A:property:error"],
            _ => &["A:property:Row:before", "B:property:Row:before", "B:property:ok", "A:property:ok"],
        };
        assert_eq!(*calls.lock().unwrap(), expected);
        assert_eq!(statement.exception_count(), match mode {2=>2,3=>1,_=>0});
        // 读取真实当前值验证下游是否被调用，不从 Filter 的返回值推断驱动状态。
        if mode == 1 || mode == 2 {
            assert_eq!(rows.int(ColumnIndex::new(1).unwrap()), Err(RdbcError::InvalidArgument("result set is not on a row".to_owned())));
            assert!(rows.next().await.unwrap());
            assert_eq!(rows.int(ColumnIndex::new(1).unwrap()).unwrap(), 10);
        } else {
            assert_eq!(rows.int(ColumnIndex::new(1).unwrap()).unwrap(), 20);
        }
        rows.close().await.unwrap();
        calls.lock().unwrap().clear();
        assert_eq!(rows.row(), Err(RdbcError::ConnectionDiscarded));
        assert_eq!(rows.navigate(ResultSetNavigation::Last).await, Err(RdbcError::ConnectionDiscarded));
        assert!(rows.is_closed().unwrap());
        assert!(calls.lock().unwrap().is_empty());
        statement.close().await.unwrap();
        connection.close().await.unwrap();
    }
}

/// 取消惰性导航不轮询后续Filter、不移动真实游标，且原资源仍可继续使用。
#[tokio::test]
async fn task6_cancelled_navigation_preserves_cursor_and_rejects_expired_lease() {
    use crate::core::{FilterChain, RdbcError};
    use druid_rdbc::result_set::{ResultSetNavigation, ResultSetExt};
    use ordered_navigation_property_filter::OrderedNavigationPropertyFilter;
    use std::sync::Mutex;
    let calls = Arc::new(Mutex::new(Vec::new()));
    let mut filters = FilterChain::new();
    for name in ["A","B","C"] {
        filters.add_result_set(Arc::new(OrderedNavigationPropertyFilter {
            name, calls: Arc::clone(&calls), mode: if name=="B" {4} else {0},
        }));
    }
    let factory = ToastyConnectionFactory::new("sqlite::memory:").await.unwrap();
    let (mut statement, mut connection) = task6_sqlite_metadata_statement(&factory, 190, "cancelled-navigation", Arc::new(filters)).await;
    let mut rows = statement.execute_query("SELECT 10 AS value UNION ALL SELECT 20").await.unwrap();
    let mut future = rows.navigate(ResultSetNavigation::Last);
    assert!(matches!(futures::poll!(&mut future), std::task::Poll::Pending));
    drop(future);
    assert_eq!(*calls.lock().unwrap(), ["A:nav:Last:before","B:nav:Last:before"]);
    assert_eq!(statement.exception_count(),0);
    assert!(rows.next().await.unwrap());
    assert_eq!(rows.int(ColumnIndex::new(1).unwrap()).unwrap(),10);
    calls.lock().unwrap().clear();
    statement.inner.lease_active.store(false,Ordering::Release);
    assert_eq!(rows.row(),Err(RdbcError::ConnectionDiscarded));
    assert_eq!(rows.navigate(ResultSetNavigation::Last).await,Err(RdbcError::ConnectionDiscarded));
    assert!(calls.lock().unwrap().is_empty());
    assert_eq!(statement.exception_count(),0);
    statement.inner.lease_active.store(true,Ordering::Release);
    rows.close().await.unwrap();
    statement.close().await.unwrap();
    connection.close().await.unwrap();
}
