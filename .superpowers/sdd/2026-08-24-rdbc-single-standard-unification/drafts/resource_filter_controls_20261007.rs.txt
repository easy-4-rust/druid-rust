/// 十三资源家族的原生重载身份、ABC顺序、B短路/错误及共享游标。
#[tokio::test]
async fn task6_resource_filters_preserve_native_overloads_order_and_stream_identity() {
    use crate::core::{FilterChain, RdbcError};
    use druid_rdbc::resource::{RdbcInputStream, RdbcReader};
    use druid_rdbc::result_set::{ResultSetExt, ResultSetRead, ResultSetStreamKind};
    use ordered_resource_filter::OrderedResourceFilter;
    use resource_read_probe::ResourceReadProbe;
    use std::sync::Mutex;
    let requests = [ResultSetRead::Ref, ResultSetRead::Blob, ResultSetRead::Clob, ResultSetRead::Array, ResultSetRead::Url, ResultSetRead::RowId, ResultSetRead::NClob, ResultSetRead::SqlXml];
    for mode in [0, 1, 2] {
        let log = Arc::new(Mutex::new(Vec::new()));
        let stream = RdbcInputStream::from_bytes([1,2,3]);
        let reader = RdbcReader::from_string("迁移");
        let mut filters = FilterChain::new();
        for name in ["A", "B", "C"] {
            filters.add_result_set(Arc::new(OrderedResourceFilter { name, mode: if name == "B" { mode } else { 0 }, calls: Arc::clone(&log), stream: stream.clone(), reader: reader.clone() }));
        }
        let (mut statement, mut connection) = task6_sqlite_diagnostic_statement_with_filters(Some(Arc::new(filters))).await;
        let admission = statement.inner.begin_standard_admission().unwrap();
        let mut rows = DruidPooledResultSet::new_standard(Arc::clone(&statement.inner), Box::new(ResourceReadProbe { calls: Arc::clone(&log), stream: stream.clone(), reader: reader.clone(), closed: false }), &admission).await.unwrap();
        drop(admission);
        macro_rules! check_read {
            ($family:expr, $target:expr, $read:expr) => {{
                log.lock().unwrap().clear();
                let result: Result<bool, RdbcError> = $read;
                let expected = if mode == 2 { Err(RdbcError::DriverError("ordered resource failure".to_owned())) } else { Ok(true) };
                assert_eq!(result, expected);
                let expected_log = if mode == 0 {
                    vec!["A:before".to_owned(), "B:before".to_owned(), "C:before".to_owned(), format!("raw:{}:{}", $family, $target), "C:ok".to_owned(), "B:ok".to_owned(), "A:ok".to_owned()]
                } else {
                    let status = if mode == 2 { "error" } else { "ok" };
                    vec!["A:before".to_owned(), "B:before".to_owned(), format!("B:{status}"), format!("A:{status}")]
                };
                assert_eq!(*log.lock().unwrap(), expected_log);
            }};
        }
        for request in &requests {
            let family = format!("resource:{request:?}");
            check_read!(&family, "index:7", rows.read_object(ColumnIndex::new(7).unwrap(), request).map(|value| value.is_null()));
            check_read!(&family, "label:virtual", rows.read_object_by_label("virtual", request).map(|value| value.is_null()));
        }
        for kind in [ResultSetStreamKind::Ascii, ResultSetStreamKind::Unicode, ResultSetStreamKind::Binary] {
            let family = format!("stream:{kind:?}");
            check_read!(&family, "index:7", rows.input_stream(ColumnIndex::new(7).unwrap(), kind).map(|value| value.is_some()));
            check_read!(&family, "label:virtual", rows.input_stream_by_label("virtual", kind).map(|value| value.is_some()));
        }
        for national in [false, true] {
            let family = format!("reader:{national}");
            check_read!(&family, "index:7", rows.reader(ColumnIndex::new(7).unwrap(), national).map(|value| value.is_some()));
            check_read!(&family, "label:virtual", rows.reader_by_label("virtual", national).map(|value| value.is_some()));
        }
        assert_eq!(statement.exception_count(), if mode == 2 { 26 } else { 0 });
        if mode == 0 {
            // 默认委托返回同一原生游标；取得句柄不消费，两个重载可接续读取。
            let indexed = rows.ascii_stream(ColumnIndex::new(7).unwrap()).unwrap().unwrap();
            let labeled = rows.ascii_stream_by_label("virtual").unwrap().unwrap();
            let mut first = [0_u8; 1];
            assert_eq!(indexed.read(&mut first).unwrap(), 1);
            assert_eq!(first, [1]);
            assert_eq!(labeled.read_to_end().unwrap(), [2,3]);
            let indexed = rows.character_stream(ColumnIndex::new(7).unwrap()).unwrap().unwrap();
            let labeled = rows.character_stream_by_label("virtual").unwrap().unwrap();
            let mut first = [0_u16; 1];
            assert_eq!(indexed.read_utf16(&mut first).unwrap(), 1);
            assert_eq!(first[0], '迁' as u16);
            assert_eq!(labeled.read_to_string().unwrap(), "移");
            macro_rules! native_error {
                ($family:expr, $read:expr) => {{
                    log.lock().unwrap().clear();
                    let result: Result<bool,RdbcError> = $read;
                    assert_eq!(result, Err(RdbcError::DriverError("native resource failure".to_owned())));
                    assert_eq!(*log.lock().unwrap(), vec!["A:before".to_owned(), "B:before".to_owned(), "C:before".to_owned(), format!("raw:{}:label:fail", $family), "C:error".to_owned(), "B:error".to_owned(), "A:error".to_owned()]);
                }};
            }
            for request in &requests { native_error!(format!("resource:{request:?}"), rows.read_object_by_label("fail", request).map(|value| value.is_null())); }
            for kind in [ResultSetStreamKind::Ascii, ResultSetStreamKind::Unicode, ResultSetStreamKind::Binary] { native_error!(format!("stream:{kind:?}"), rows.input_stream_by_label("fail", kind).map(|value| value.is_some())); }
            for national in [false,true] { native_error!(format!("reader:{national}"), rows.reader_by_label("fail", national).map(|value| value.is_some())); }
            assert_eq!(statement.exception_count(), 13);
        }
        rows.close().await.unwrap();
        log.lock().unwrap().clear();
        assert_eq!(rows.binary_stream_by_label("virtual"), Err(RdbcError::ConnectionDiscarded));
        assert_eq!(rows.character_stream_by_label("virtual"), Err(RdbcError::ConnectionDiscarded));
        assert!(log.lock().unwrap().is_empty());
        statement.close().await.unwrap();
        connection.close().await.unwrap();
    }
}

/// 资源短路仍服从隐藏列、关闭与旧租约治理，失败只向实际创建语句记录一次。
#[tokio::test]
async fn task6_resource_filters_obey_column_visibility_close_and_expired_lease() {
    use crate::core::{FilterChain, RdbcError};
    use druid_rdbc::resource::{RdbcInputStream, RdbcReader};
    use druid_rdbc::result_set::{ResultSetRead, ResultSetStreamKind};
    use resource_short_circuit_filter::ResourceShortCircuitFilter;
    use std::sync::Mutex;
    let requests = [ResultSetRead::Ref,ResultSetRead::Blob,ResultSetRead::Clob,ResultSetRead::Array,ResultSetRead::Url,ResultSetRead::RowId,ResultSetRead::NClob,ResultSetRead::SqlXml];
    for expired in [false,true] {
        let log = Arc::new(Mutex::new(Vec::new()));
        let mut filters = FilterChain::new();
        filters.add_result_set(Arc::new(ResourceShortCircuitFilter { calls: Arc::clone(&log), stream: RdbcInputStream::from_bytes([1,2,3]), reader: RdbcReader::from_string("迁移") }));
        let (mut rows, mut statement, mut connection) = task6_sqlite_diagnostic_cursor_with_sql_and_filters("SELECT 1 AS value", Some(Arc::new(filters))).await;
        let error = if expired {
            statement.inner.lease_active.store(false, Ordering::Release);
            RdbcError::ConnectionDiscarded
        } else {
            rows.set_hidden_columns(Some(vec![1]));
            RdbcError::InvalidArgument("column is hidden".to_owned())
        };
        for request in &requests {
            assert_eq!(rows.read_object(ColumnIndex::new(1).unwrap(), request), Err(error.clone()));
            assert_eq!(rows.read_object_by_label("value", request), Err(error.clone()));
        }
        for kind in [ResultSetStreamKind::Ascii,ResultSetStreamKind::Unicode,ResultSetStreamKind::Binary] {
            assert_eq!(rows.input_stream(ColumnIndex::new(1).unwrap(), kind), Err(error.clone()));
            assert_eq!(rows.input_stream_by_label("value", kind), Err(error.clone()));
        }
        for national in [false,true] {
            assert_eq!(rows.reader(ColumnIndex::new(1).unwrap(), national), Err(error.clone()));
            assert_eq!(rows.reader_by_label("value", national), Err(error.clone()));
        }
        assert!(log.lock().unwrap().is_empty());
        assert_eq!(statement.exception_count(), if expired { 0 } else { 26 });
        rows.close().await.unwrap();
        assert_eq!(rows.input_stream_by_label("value", ResultSetStreamKind::Ascii), Err(RdbcError::ConnectionDiscarded));
        assert_eq!(rows.reader_by_label("value", false), Err(RdbcError::ConnectionDiscarded));
        assert!(log.lock().unwrap().is_empty());
        assert_eq!(statement.exception_count(), if expired { 0 } else { 26 });
        statement.inner.lease_active.store(true, Ordering::Release);
        statement.close().await.unwrap();
        connection.close().await.unwrap();
    }
}
