/// 原生列名资源读取保持NULL、定位、未知列与关闭错误的优先级。
/// 对应 Java: ResultSet#getRef/getBlob/getClob/getArray/getURL/getRowId/getNClob/getSQLXML。
#[tokio::test]
async fn native_resource_labels_preserve_null_and_error_priority() {
    use druid_rdbc::result_set::ResultSetRead;
    use druid_rdbc::types::RdbcObject;
    let factory = ToastyConnectionFactory::new("sqlite::memory:").await.unwrap();
    let mut connection = factory.create().await.unwrap();
    let mut statement = connection.create_statement(StatementOptions::default()).await.unwrap();
    let mut rows = statement.execute_query("SELECT 1 AS id, NULL AS empty_resource").await.unwrap();
    let requests = [
        (ResultSetRead::Ref, "result_set_ref_by_label"),
        (ResultSetRead::Blob, "result_set_blob_by_label"),
        (ResultSetRead::Clob, "result_set_clob_by_label"),
        (ResultSetRead::Array, "result_set_array_by_label"),
        (ResultSetRead::Url, "result_set_url_by_label"),
        (ResultSetRead::RowId, "result_set_row_id_by_label"),
        (ResultSetRead::NClob, "result_set_n_clob_by_label"),
        (ResultSetRead::SqlXml, "result_set_sql_xml_by_label"),
    ];
    for (request, _) in &requests {
        assert_eq!(rows.read_object_by_label("id", request), Err(RdbcError::Other("result set cursor is not positioned on a row".to_owned())));
    }
    assert!(rows.next().await.unwrap());
    for (request, operation) in &requests {
        assert_eq!(rows.read_object_by_label("id", request), Err(RdbcError::UnsupportedOperation { operation }));
        assert_eq!(rows.read_object_by_label("empty_resource", request), Ok(RdbcObject::Scalar(Value::Null)));
        assert_eq!(rows.was_null(), Ok(true));
        assert_eq!(rows.read_object_by_label("missing", request), Err(RdbcError::InvalidArgument("unknown column: missing".to_owned())));
    }
    rows.close().await.unwrap();
    for (request, _) in &requests {
        assert_eq!(rows.read_object_by_label("missing", request), Err(RdbcError::ConnectionDiscarded));
    }
    statement.close().await.unwrap();
    connection.close().await.unwrap();
}
