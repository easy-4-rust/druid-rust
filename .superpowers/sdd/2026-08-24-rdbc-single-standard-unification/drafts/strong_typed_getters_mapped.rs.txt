/// 保留完整 Decimal、日期、时间、时间戳及 Calendar 重载与 SQL NULL 合同。
/// 对应 Java: DruidPooledResultSet#getBigDecimal/getDate/getTime/getTimestamp。
#[tokio::test]
async fn task6_strong_typed_getters_preserve_all_decimal_temporal_and_calendar_overloads() {
    use bigdecimal::BigDecimal;
    use chrono::{NaiveDate, NaiveDateTime, NaiveTime};
    use druid_rdbc::extensions::BufferedRowSet;
    use druid_rdbc::metadata::ResultSetColumnType;
    use druid_rdbc::result_set::ResultSetExt;
    use druid_rdbc::types::{RdbcCalendar, RdbcCalendarArgument, Row, Value};
    use std::str::FromStr;
    let (mut statement, mut connection) = task6_sqlite_diagnostic_statement_with_filters(None).await;
    let decimal = BigDecimal::from_str("1234567890.45").unwrap();
    let date = NaiveDate::from_ymd_opt(2026, 7, 29).unwrap();
    let time = NaiveTime::from_hms_nano_opt(13, 14, 15, 123_456_789).unwrap();
    let timestamp = NaiveDateTime::new(date, time);
    let physical: Box<dyn ResultSet> = Box::new(BufferedRowSet::with_column_labels(
        vec![Row::new(vec![
            Value::Decimal(decimal.clone()),
            Value::Date(date),
            Value::Time(time),
            Value::Timestamp(timestamp),
            Value::Null,
        ])],
        vec![
            "amount".to_string(),
            "event_date".to_string(),
            "event_time".to_string(),
            "event_at".to_string(),
            "optional_amount".to_string(),
        ],
    ));
    let admission = statement.inner.begin_standard_admission().unwrap();
    let mut result_set = DruidPooledResultSet::new_standard(Arc::clone(&statement.inner), physical, &admission).await.unwrap();
    drop(admission);
    assert!(result_set.next().await.unwrap());

    let meta_data = result_set.meta_data().unwrap();
    assert_eq!(
        meta_data.column_type(ColumnIndex::new(1).unwrap()).unwrap(),
        ResultSetColumnType::Decimal
    );
    assert_eq!(meta_data.column_type(ColumnIndex::new(2).unwrap()).unwrap(), ResultSetColumnType::Date);
    assert_eq!(meta_data.column_type(ColumnIndex::new(3).unwrap()).unwrap(), ResultSetColumnType::Time);
    assert_eq!(
        meta_data.column_type(ColumnIndex::new(4).unwrap()).unwrap(),
        ResultSetColumnType::Timestamp
    );

    assert_eq!(
        result_set.big_decimal(ColumnIndex::new(1).unwrap()).unwrap(),
        Some(decimal.clone())
    );
    assert_eq!(
        result_set
            .big_decimal_by_label("amount")
            .unwrap(),
        Some(decimal.clone())
    );
    assert_eq!(
        result_set
            .big_decimal_with_scale(ColumnIndex::new(1).unwrap(), 2)
            .unwrap(),
        Some(decimal.clone())
    );
    assert_eq!(
        result_set
            .big_decimal_by_label_with_scale("amount", 2)
            .unwrap(),
        Some(decimal)
    );

    let shanghai = RdbcCalendar::new("Asia/Shanghai").unwrap();
    assert_eq!(result_set.date(ColumnIndex::new(2).unwrap(), &RdbcCalendarArgument::Unspecified).unwrap(), Some(date));
    assert_eq!(
        result_set
            .date_by_label("event_date", &RdbcCalendarArgument::Unspecified)
            .unwrap(),
        Some(date)
    );
    assert_eq!(
        result_set
            .date(ColumnIndex::new(2).unwrap(), &RdbcCalendarArgument::Specified(Some(shanghai.clone())))
            .unwrap(),
        Some(date)
    );
    assert_eq!(
        result_set
            .date_by_label("event_date", &RdbcCalendarArgument::Specified(None))
            .unwrap(),
        Some(date)
    );

    assert_eq!(result_set.time(ColumnIndex::new(3).unwrap(), &RdbcCalendarArgument::Unspecified).unwrap(), Some(time));
    assert_eq!(
        result_set
            .time_by_label("event_time", &RdbcCalendarArgument::Unspecified)
            .unwrap(),
        Some(time)
    );
    assert_eq!(
        result_set
            .time(ColumnIndex::new(3).unwrap(), &RdbcCalendarArgument::Specified(None))
            .unwrap(),
        Some(time)
    );
    assert_eq!(
        result_set
            .time_by_label("event_time", &RdbcCalendarArgument::Specified(Some(shanghai.clone())))
            .unwrap(),
        Some(time)
    );

    assert_eq!(
        result_set.timestamp(ColumnIndex::new(4).unwrap(), &RdbcCalendarArgument::Unspecified).unwrap(),
        Some(timestamp)
    );
    assert_eq!(
        result_set
            .timestamp_by_label("event_at", &RdbcCalendarArgument::Unspecified)
            .unwrap(),
        Some(timestamp)
    );
    assert_eq!(
        result_set
            .timestamp(ColumnIndex::new(4).unwrap(), &RdbcCalendarArgument::Specified(Some(shanghai)))
            .unwrap(),
        Some(timestamp)
    );
    assert_eq!(
        result_set
            .timestamp_by_label("event_at", &RdbcCalendarArgument::Specified(None))
            .unwrap(),
        Some(timestamp)
    );

    assert_eq!(
        result_set
            .big_decimal_by_label("optional_amount")
            .unwrap(),
        None
    );
    assert!(result_set.was_null().unwrap());
    result_set.close().await.unwrap();
    statement.close().await.unwrap();
    connection.close().await.unwrap();
}
